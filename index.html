The program below asks the user to input 12 numbers between -100 and 100 and stores those numbers in an array, then it displays the numbers in the Original order and in Reversed order. It then calculates the Sum of all the numbers, the Highest number, the Lowest number, and the Average.

 
import java.util.Scanner;

public class ArrayReversal {
    public static void main(String[] args) {
        /* Kian Josh DC. Bumanlag

        Scanner userinput = new Scanner(System.in);
        int[] Integers = new int[12]; // Declaration of Array with a size that can hold 12 integers
        
        /*   V Input and Validation V  */
        System.out.println("\n Enter 12 Numbers between -100 and 100:");
        System.out.println();
        for (int n = 0; n < 12; n++) {
            while (true) {
                System.out.print("Enter a number " + n + ": ");
                if (userinput.hasNextInt()) {
                    int number = userinput.nextInt();
                    if (number >= -100 && number <= 100) {
                        Integers[n] = number;
                        break;
                    } else {
                        System.out.println("Error: Numbers should only be between -100 & 100.");
                    }
                } else {
                    System.out.println("Error: Please enter a valid Number.");
                    userinput.next();                   
                }
            }
        }
        
        /*   V Original Sequence V  */
        System.out.println();
        System.out.println("\n Orginal Sequence ");
        System.out.println();
        for (int n = 0; n < 12; n++) { 
        System.out.println("Number " + n + ": " + Integers[n]);
        }
        
        /*   V Reversed Sequence V  */
        System.out.println();
        System.out.println("\n Reversed Sequence ");
        System.out.println();
       for (int n = 11; n >= 0; n--) {
       	System.out.println("Number " + n + ": " + Integers[n]);
       }
       
       /*  V Results: Sum, Highest, Lowest, and Average V  */
       int S = 0;
       int H = Integers[0];
       int L = Integers[0];
       
       for (int n = 0; n < 12; n++) {
       	S += Integers[n];
       	
       	if (Integers[n] > H) {
       		H = Integers[n];
       	}
       	if (Integers[n] < L) {
       		L = Integers[n];
       	}
       }
       
       double A = (double) S / 12;
       
       System.out.println();
       System.out.println("\n Results ");
       System.out.println("Sum: " + S);
       System.out.println("Average: " + A);
       System.out.println("Highest: " + H);
       System.out.println("Lowest: " + L);
       
    }
}
